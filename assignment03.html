# 2) Parameters 
state_abbr <- "TX"
county_name <- "Dallas"
geo_level  <- "tract"   # options: state, county, tract, block group
my_vars    <- c(income = "B19013_001", poverty = "B17001_002")
year_acs   <- 2023
survey     <- "acs5"

# 3) Get Data
acs <- get_acs(
  geography = geo_level,
  variables = my_vars,
  state = state_abbr,
  county = county_name,
  year = year_acs,
  survey = survey,
  geometry = TRUE
)
Getting data from the 2019-2023 5-year ACS
# 4) Change to Wide format for convenience
acs_wide <- acs |>
  tidyr::pivot_wider(
    id_cols = c("GEOID", "NAME", "geometry"),
    names_from = "variable",
    values_from = c("estimate", "moe")
  )

# 5) Map 
ggplot(acs_wide) +
  geom_sf(aes(fill = estimate_income), color = NA) +

  labs(title = paste0("ACS ", year_acs, " 5-year: Median Income — ", state_abbr, " (", county_name, ")"),
       caption = "Source: U.S. Census Bureau via tidycensus") +
  theme_minimal() +
  theme(plot.title = element_text(size = 14, face = "bold"),

3rd Deliverable: Interpret map and table data

  • Highest Median Income: The map visualization help to identify that the region with the highest median income (> $250,000) in the county of Dallas is located in the central/slightly north area (known as ‘Highland Park’).

  • Lowest Median Income: The maps also displays that the areas with lowest median income is located in the central/slightly south east area of Dallas county.

  • Average: Most census tracts in the county of Dallas exhibit a median income between $50k to $150k.
